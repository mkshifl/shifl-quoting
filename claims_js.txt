function getAllShipments(){
  const drayage = (S.quotes||[]).map((q,i)=>({
    _idx:i,_type:'drayage',_label:'Drayage',
    id: q.shiflRef||q.id||('DRY-'+i),
    customer: q.customer||'—',
    origin: q.port||'Newark',
    dest: q.destination||q.zip||'—',
    carrier: q.carrier||'—',
    status: q.status||'—',
    date: q.date||q.created_at||'',
    amount: q.customerRates?.total||0
  }));
  const freight = (window._fqHistory||[]).map((q,i)=>({
    _idx:i,_type:'freight',_label: q.fqMode||'Freight',
    id: q.quoteNum||q.shiflRef||('FQ-'+i),
    customer: q.customer||'—',
    origin: q.pickupZip||q.originCity||'—',
    dest: q.deliveryZip||q.destCity||'—',
    carrier: q.carrier||'—',
    status: q.status||'—',
    date: q.date||q.created_at||'',
    amount: q.customerRates?.total||0
  }));
  const air = (window._aqHistory||[]).map((q,i)=>({
    _idx:i,_type:'air',_label:'Air',
    id: q.quoteNum||('AQ-'+i),
    customer: q.customer||'—',
    origin: q.originAirport||'—',
    dest: q.destAirport||'—',
    carrier: q.carrier||'—',
    status: q.status||'—',
    date: q.date||q.created_at||'',
    amount: q.customerRates?.total||0
  }));
  return [...drayage,...freight,...air].sort((a,b)=>new Date(b.date)-new Date(a.date));
}

function getAllClaims(){
  const all = [];
  try{
    const keys = Object.keys(localStorage).filter(k=>k.startsWith('tms_claims_'));
    keys.forEach(k=>{
      const claims = JSON.parse(localStorage.getItem(k)||'[]');
      claims.forEach(c=>all.push(c));
    });
    // Also check new unified claims store
    const unified = JSON.parse(localStorage.getItem('shifl_claims')||'[]');
    unified.forEach(c=>{ if(!all.find(x=>x.id===c.id)) all.push(c); });
  }catch(e){}
  return all.sort((a,b)=>new Date(b.filedAt||b.ts||0)-new Date(a.filedAt||a.ts||0));
}

function saveClaim(claim){
  const all = JSON.parse(localStorage.getItem('shifl_claims')||'[]');
  const idx = all.findIndex(c=>c.id===claim.id);
  if(idx>=0) all[idx]=claim;
  else all.unshift(claim);
  localStorage.setItem('shifl_claims',JSON.stringify(all));
  // Update badge
  const open = all.filter(c=>c.status!=='resolved').length;
  const badge = document.getElementById('claims-badge');
  if(badge){ badge.textContent=open; badge.style.display=open>0?'inline':'none'; }
}

function claimStatusPill(key){
  const s = CLAIM_STATUSES.find(x=>x.key===key)||CLAIM_STATUSES[0];
  return `<span style="padding:3px 10px;border-radius:99px;font-size:10px;font-weight:700;background:${s.bg};color:${s.color}">${s.label}</span>`;
}

let _claimStep = 1;
let _claimFilter = 'all';

function openNewClaimWizard(){
  _claimDraft = {};

}

function selectClaimShipment(id, type, label){
  _claimDraft.shipmentId = id;
  _claimDraft.type_raw = type;
  _claimDraft.mode = label;
  renderClaimWizardModal();
}

function claimStep2(){
  if(!_claimDraft.shipmentId) return;
  _claimStep = 2;
  renderClaimWizardModal();
}

function claimAddFiles(){
  var input = document.createElement('input');
  input.type = 'file';
  input.multiple = true;
  input.accept = '.pdf,.jpg,.jpeg,.png,.doc,.docx';
  input.onchange = function(){
    _claimDraft.files = _claimDraft.files||[];
    Array.from(input.files).forEach(function(f){
      var reader = new FileReader();
      reader.onload = function(e){
        _claimDraft.files.push({name:f.name,data:e.target.result});
        var el = document.getElementById('cl-files-list');
        if(el) el.innerHTML = _claimDraft.files.map(function(fi){return '<div style="display:inline-block;background:#f0f6ff;border-radius:5px;padding:2px 8px;margin:2px;font-size:11px">📎 '+fi.name+'</div>';}).join('');
      };
      reader.readAsDataURL(f);
    });
  };
  input.click();
}

function submitClaim(){
  var type = (document.getElementById('cl-type')||{}).value||'';
  var notes = ((document.getElementById('cl-notes')||{}).value||'').trim();
  var amount = parseFloat((document.getElementById('cl-amount')||{}).value)||0;
  var against = (document.getElementById('cl-against')||{}).value||'carrier';
  if(!type){ alert('Please select a claim type.'); return; }
  if(!notes){ alert('Please describe the claim.'); return; }

  var claim = {
    id: 'CLM-'+Date.now(),
    shipmentId: _claimDraft.shipmentId,
    mode: _claimDraft.mode,
    customer: _claimDraft.customer||'—',
    type: type,
    notes: notes,
    amount: amount,
    against: against,
    files: _claimDraft.files||[],
    status: 'filed',
    filedAt: new Date().toISOString(),
    updates: [{status:'filed',note:'Claim filed.',ts:new Date().toISOString()}]
  };

  saveClaim(claim);
  closeModal();
  showToast('⚖️ Claim '+claim.id+' filed successfully!');
  setView('claims'); render();
}

function addClaimUpdate(claimId){
  const newStatus = (document.getElementById('cl-new-status')||{}).value||'filed';
  const note = ((document.getElementById('cl-update-note')||{}).value||'').trim();
  const all = JSON.parse(localStorage.getItem('shifl_claims')||'[]');
  const cl = all.find(c=>c.id===claimId);
  if(!cl) return;
  cl.status = newStatus;
  cl.updates = cl.updates||[];
  cl.updates.push({status:newStatus,note:note,ts:new Date().toISOString()});
  localStorage.setItem('shifl_claims',JSON.stringify(all));
  showToast('Claim updated → '+( CLAIM_STATUSES.find(s=>s.key===newStatus)||{label:newStatus}).label);
  closeModal();
  renderClaims();
}

function updateClaimStatus(claimId){
  openClaimDetail(claimId);
}

    const badge = document.getElementById('claims-badge');
    if(badge){ badge.textContent=open; badge.style.display=open>0?'inline':'none'; }
  }catch(e){}
}
